// A switch whose last case sits after the function's final `jr ra` looks to
// splat like a new function starting there. The jump table says otherwise:
// every entry is a case of the one switch.

// jtbl_800A7E60 sends 4 of its 13 cases to 0x002682FC, after the jr ra.
func_002682CC = 0x002682CC; // type:func size:0x38

// that they reach by `j`. splat stopped the table at 4 and read the fifth
// entry as a .double.
func_002640DC = 0x002640DC; // type:func size:0x94
